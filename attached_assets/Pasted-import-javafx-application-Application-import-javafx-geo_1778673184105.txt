import javafx.application.Application;
import javafx.geometry.Pos;
import javafx.scene.Scene;
import javafx.scene.control.*;
import javafx.scene.layout.*;
import javafx.scene.paint.Color;
import javafx.scene.text.Font;
import javafx.stage.Stage;
import javafx.animation.*;
import javafx.util.Duration;
import java.util.*;

public class Main extends Application {
    private int currentQuestion = 0;
    private int score = 0;
    private List<Question> questions = new ArrayList<>();
    private Label questionLabel;
    private ToggleGroup optionsGroup;
    private VBox root;
    private Button nextButton;
    private Button submitButton;

    @Override
    public void start(Stage stage) {
        stage.setTitle("CompTIA Tryfecta Combo Exam Trainer");
        stage.setWidth(800);
        stage.setHeight(600);

        // Metallic blue gradient background
        root = new VBox(20);
        root.setAlignment(Pos.CENTER);
        root.setStyle("-fx-background-color: linear-gradient(to bottom right, #001f3f, #0074D9);");

        questionLabel = new Label();
        questionLabel.setFont(Font.font("Segoe UI", 22));
        questionLabel.setTextFill(Color.SILVER);

        optionsGroup = new ToggleGroup();

        nextButton = new Button("Next");
        nextButton.setFont(Font.font("Segoe UI", 16));
        nextButton.setStyle("-fx-background-color: #0074D9; -fx-text-fill: white; -fx-background-radius: 10;");
        nextButton.setOnAction(e -> nextQuestion(stage));

        submitButton = new Button("Submit");
        submitButton.setFont(Font.font("Segoe UI", 16));
        submitButton.setStyle("-fx-background-color: #2ECC40; -fx-text-fill: white; -fx-background-radius: 10;");
        submitButton.setOnAction(e -> showResults(stage));

        loadQuestions();
        showQuestion(stage);

        Scene scene = new Scene(root);
        stage.setScene(scene);
        stage.show();
    }

    private void loadQuestions() {
        questions.add(new Question("Which port does HTTP use?", new String[]{"80", "443", "21", "25"}, 0));
        questions.add(new Question("Which layer of the OSI model handles routing?", new String[]{"Network", "Transport", "Data Link", "Application"}, 0));
        questions.add(new Question("What is the primary goal of Security+?", new String[]{"Network setup", "Hardware repair", "Cybersecurity fundamentals", "Cloud management"}, 2));
        // Add more questions here...
    }

    private void showQuestion(Stage stage) {
        root.getChildren().clear();
        Question q = questions.get(currentQuestion);
        questionLabel.setText(q.text);

        VBox optionsBox = new VBox(10);
        optionsBox.setAlignment(Pos.CENTER);
        optionsGroup.getToggles().clear();

        for (int i = 0; i < q.options.length; i++) {
            RadioButton rb = new RadioButton(q.options[i]);
            rb.setFont(Font.font("Segoe UI", 18));
            rb.setTextFill(Color.WHITE);
            rb.setToggleGroup(optionsGroup);
            optionsBox.getChildren().add(rb);
        }

        root.getChildren().addAll(questionLabel, optionsBox, nextButton, submitButton);

        // Randomized transition effect
        int effectType = new Random().nextInt(3);
        Transition transition;
        switch (effectType) {
            case 0:
                transition = new FadeTransition(Duration.millis(700), root);
                ((FadeTransition) transition).setFromValue(0);
                ((FadeTransition) transition).setToValue(1);
                break;
            case 1:
                transition = new TranslateTransition(Duration.millis(700), root);
                ((TranslateTransition) transition).setFromX(-800);
                ((TranslateTransition) transition).setToX(0);
                break;
            default:
                transition = new ScaleTransition(Duration.millis(700), root);
                ((ScaleTransition) transition).setFromX(0.5);
                ((ScaleTransition) transition).setToX(1);
                ((ScaleTransition) transition).setFromY(0.5);
                ((ScaleTransition) transition).setToY(1);
                break;
        }
        transition.play();
    }

    private void nextQuestion(Stage stage) {
        RadioButton selected = (RadioButton) optionsGroup.getSelectedToggle();
        if (selected != null) {
            Question q = questions.get(currentQuestion);
            if (selected.getText().equals(q.options[q.correctIndex])) {
                score++;
                root.setStyle("-fx-background-color: linear-gradient(to bottom right, #001f3f, #2ECC40);");
            } else {
                root.setStyle("-fx-background-color: linear-gradient(to bottom right, #001f3f, #FF4136);");
            }
        }

        currentQuestion++;
        if (currentQuestion < questions.size()) {
            showQuestion(stage);
        } else {
            showResults(stage);
        }
    }

    private void showResults(Stage stage) {
        root.getChildren().clear();
        Label resultLabel = new Label("Your Score: " + score + " / " + questions.size());
        resultLabel.setFont(Font.font("Segoe UI", 28));
        resultLabel.setTextFill(Color.SILVER);
        root.getChildren().add(resultLabel);
    }

    public static void main(String[] args) {
        launch(args);
    }

    class Question {
        String text;
        String[] options;
        int correctIndex;

        Question(String text, String[] options, int correctIndex) {
            this.text = text;
            this.options = options;
            this.correctIndex = correctIndex;
        }
    }
}
